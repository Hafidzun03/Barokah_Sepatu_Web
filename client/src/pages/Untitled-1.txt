  // Simple drawer toggle interactions
  function openDrawer(customerId) {

    backdrop.classList.remove('opacity-0');
    backdrop.classList.add('opacity-100');
    panel.classList.remove('translate-x-full');

  function closeDrawer() {
    const drawer = document.getElementById('customerDrawer');
    const backdrop = document.getElementById('drawerBackdrop');
    const panel = document.getElementById('drawerPanel');
    
    backdrop.classList.add('opacity-0');
    backdrop.classList.remove('opacity-100');
    panel.classList.add('translate-x-full');
    setTimeout(() => {
      drawer.classList.add('pointer-events-none');
    }, 300);
  }

  function openOrderModal(customerId) {
    // Open drawer focused directly
    openDrawer(customerId);

  // Broadcast Modal controls
  const broadcastBtn = document.getElementById('btnBroadcast');
  const broadcastModal = document.getElementById('broadcastModal');

  if (broadcastBtn && broadcastModal) {
    broadcastBtn.addEventListener('click', () => {
      broadcastModal.classList.remove('pointer-events-none', 'opacity-0');
      broadcastModal.classList.add('opacity-100');
    });
  }

  function closeBroadcastModal() {
    if (broadcastModal) {
      broadcastModal.classList.add('opacity-0', 'pointer-events-none');
      broadcastModal.classList.remove('opacity-100');
    }
  }

  function confirmBroadcast() {
    alert('Notifikasi Secret Drop berhasil dijadwalkan ke 1.140 nomor VIP member!');
    closeBroadcastModal();

  // Filter toolbar client search demo
  const searchInput = document.getElementById('searchInput');
  const tableRows = document.querySelectorAll('#customerTableBody tr');

  if (searchInput) {
    searchInput.addEventListener('input', (e) => {
      const query = e.target.value.toLowerCase();
      tableRows.forEach(row => {
        const text = row.innerText.toLowerCase();
        row.style.display = text.includes(query) ? '' : 'none';
      });
    });
  }

  // Filter tab active switch

      filterTabs.forEach(t => {
        t.classList.remove('bg-primary', 'text-on-primary', 'font-bold', 'shadow-sm');
        t.classList.add('text-on-surface-variant');
      });
      tab.classList.add('bg-primary', 'text-on-primary', 'font-bold', 'shadow-sm');
      tab.classList.remove('text-on-surface-variant');
    });
  });
